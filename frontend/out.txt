19 /** 平滑滚动到页面最顶部（兼容不支持 smooth 的老浏览器） */
20 function scrollToTop() {
21   if (typeof window === 'undefined') return
22   if (typeof window.scrollTo === 'function') {
23     try {
24       window.scrollTo({ top: 0, behavior: 'smooth' })
25       return
26     } catch (e) {
27       // 部分老浏览器不支持对象参数，落到下面的降级写法
28     }
29     window.scrollTo(0, 0)
30   }
31 }

40 // 判断某项是否处于高亮：
41 // - 首页 '/' 需精确匹配，避免所有路由都被当成首页
42 // - 其它路径按前缀匹配，兼容子路由（如 /competitions/123）
43 function isActive(item) {
44   const current = route.path || '/'
45   if (item.path === '/') {
46     return current === '/'
47   }
48   return current === item.path || current.startsWith(item.path + '/')
49 }
