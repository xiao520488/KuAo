// 注意：requestId / unmounted 都是每次 setup 新建的闭包变量，
//      旧实例自增自己的 requestId 不会影响新实例的计数器，
//      因此不会出现"旧组件卸载把新组件状态搞乱"的问题。
let unmounted = false

  unmounted = true

  requestId++
})
