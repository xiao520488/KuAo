 * 签名 = 卡片 id 顺序拼接；仅当内容真的变了才重放，
 * 避免「点了备赛按钮 / 切换选中态」这类无关更新触发整列重播。
 */
function bumpListAnimation() {
  const signature = competitions.value.map((c) => c.id).join('|')
  if (signature === lastSignature) return
  lastSignature = signature
  listAnimKey.value += 1
}
