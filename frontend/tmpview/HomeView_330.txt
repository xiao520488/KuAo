    if (val) {
      keyword.value = val
      runSearch(val)
      return
    }

    // 无 q：首次挂载直接加载热门列表
    if (oldQ === undefined) {
      loadCompetitions()
      return
    }

    // 原本有 q、现在被清掉（如点 Logo 返回首页）→ 恢复正常列表
    resetToDefault()
  },
  { immediate: true }
)
</script>
