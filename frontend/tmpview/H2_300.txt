/**
 * 是否展示「查看更多」引导按钮
 * 仅在默认「热门竞赛」态下出现：
 *   - !searched   → 非 AI 检索结果态
 *   - !loading    → 骨架屏已结束
 *   - !searching  → 不在检索进行中
 *   - 列表非空    → 避免空态下引导到列表页显得突兀
 */
const showViewMore = computed(
  () => !searched.value && !loading.value && !searching.value && competitions.value.length > 0
)
